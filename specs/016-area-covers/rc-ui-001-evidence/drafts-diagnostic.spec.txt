import {test,expect} from '@playwright/test';
import {writeFile} from 'node:fs/promises';
import {signSessionToken} from '../src/app/api/server/auth/session';
import {ENVIRONMENTAL_FIXTURES as F,setupEnvironmentalFixtures,cleanupEnvironmentalFixtures,countEnvironmentalFixtures} from '../tests/fixtures/environmental-data-fixtures';
test.beforeAll(async()=>setupEnvironmentalFixtures(process.env));
test.afterAll(async()=>{await cleanupEnvironmentalFixtures(process.env);expect(Object.values(await countEnvironmentalFixtures(process.env)).every(n=>n===0)).toBe(true);});
const base=`/dashboard/laboratories/${F.laboratoryIds[0]}/areas/${F.areaIds[0]}/collections/${F.collectionIds[0]}/environmental-data`;
test('diagnostic: capture selection and first save across delayed hydration',async({page,context},info)=>{
 test.setTimeout(60000);
 await context.addCookies([{name:'auth_token',value:signSessionToken(F.userIds[2]),url:String(info.project.use.baseURL)}]);
 let release!:()=>void;const blocked=new Promise<void>(resolve=>{release=resolve;});
 let scripts=0;
 await page.route(url=>url.pathname.startsWith('/_next/static/')&&url.pathname.endsWith('.js'),async route=>{scripts++;await blocked;await route.continue();});
 const observations:unknown[]=[];
 const observe=async(label:string)=>{observations.push({label,values:await page.locator('select[name="water.hasSpring"],input[name="soil.infiltrationRateMmPerHour"],input[name="terrain.elevationMeters"]').evaluateAll(els=>els.map(el=>({name:el.getAttribute('name'),value:(el as HTMLInputElement).value,reactHandler:Object.keys(el).some(k=>k.startsWith('__reactProps'))}))),spring:await page.getByLabel('Há nascente').inputValue()});};
 try {
  await page.goto(base+'/new',{waitUntil:'commit'});
  await expect(page.getByLabel('Há nascente')).toBeVisible();
  await observe('before-selection');
  await page.getByLabel('Há nascente').selectOption('false');
  await observe('after-native-selection-before-hydration');
 } finally {release();}
 await page.waitForLoadState('load');
 await expect.poll(()=>page.getByLabel('Há nascente').evaluate(el=>Object.keys(el).some(k=>k.startsWith('__reactProps')))).toBe(true);
 await page.getByLabel('Taxa de infiltração').fill('0');await page.getByLabel('Elevação').fill('12.');
 await observe('after-hydration-and-number-changes');
 const request=page.waitForRequest(r=>r.url().endsWith('/draft')&&r.method()==='PUT');
 await page.getByRole('button',{name:'Salvar rascunho',exact:true}).click();
 const fields=(await request).postDataJSON().fields;
 observations.push({label:'first-save-payload',fields});
 await expect(page.getByRole('status')).toContainText('Rascunho salvo.');
 await page.reload();await observe('after-reload');
 await writeFile('.accounts-validation/drafts-hydration-diagnostic.json',JSON.stringify({kind:'synthetic-diagnostic',scripts,observations},null,2));
 expect(fields['water.hasSpring']).toEqual({kind:'editing',raw:'false'});
});
